1""" 

2Data preparation orchestrator — runs load -> EDA -> split -> clean -> 

3engineer -> SMOTE as one callable pipeline. 

4 

5Split happens BEFORE cleaning/feature-engineering are fit. This 

6ordering is the whole point: imputation medians, outlier IQR bounds, 

7and the RobustScaler are all statistics learned from data, and fitting 

8them on anything that includes the test (or validation) split leaks 

9that split's information into how the training data gets transformed. 

10Every fitted statistic here is fit once, on the train split only, and 

11then applied (not refit) to val and test -- the same discipline a 

12StandardScaler tutorial would call "fit_transform on train, transform 

13on test", just extended to the imputer and outlier-bound step too. 

14 

15Callable as a script: 

16 python -m app.ml.pipeline.prepare_data --input path/to/creditcard.csv 

17 

18Also callable programmatically (used by train.py in Phase 6, and by 

19the pipeline's own tests). 

20""" 

21

38def prepare_data( 

39 input_path: str | Path, 

40 *, 

41 apply_imbalance_handling: bool = True, 

42 smote_sampling_strategy: float = 0.5, 

43 test_size: float = 0.2, 

44 val_size: float = 0.15, 

45) -> dict[str, Any]: 

46 """ 

47 Runs the complete data preparation pipeline against a single CSV 

48 and returns everything downstream stages need: 

49 - X_train, y_train (post-SMOTE) 

50 - X_val, y_val (real distribution, no SMOTE -- for threshold selection) 

51 - X_test, y_test (real distribution, no SMOTE, touched only by 

52 the train-fitted scaler -- for final reported metrics) 

53 - the fitted RobustScaler (must be persisted alongside the model) 

54 - a combined report from every stage, for logging/auditing 

55 """ 

56 report: dict[str, Any] = {"input_path": str(input_path)} 

57 

58 # ---- 1. Load ---- 

59 df = load_transactions_csv(input_path, require_label=True) 

60 report["raw_summary"] = dataset_summary(df) 

61 

62 # ---- 2. EDA (purely descriptive on the raw data -- fits nothing, 

63 # so running it before the split doesn't leak anything) ---- 

64 report["eda"] = run_eda(df) 

65 

66 # ---- 3. Split FIRST, before any statistic is fit ---- 

67 train_raw, val_raw, test_raw = three_way_split_raw(df, test_size=test_size, val_size=val_size) 

68 report["split"] = { 

69 "train_rows": len(train_raw), 

70 "val_rows": len(val_raw), 

71 "test_rows": len(test_raw), 

72 "train_fraud_count": int(train_raw["Class"].sum()), 

73 "val_fraud_count": int(val_raw["Class"].sum()), 

74 "test_fraud_count": int(test_raw["Class"].sum()), 

75 } 

76 

77 # ---- 4. Clean: fit imputation/outlier stats on train only, reuse on val/test ---- 

78 train_clean, cleaning_report, fitted_clean_params = clean_data(train_raw) 

79 val_clean, _, _ = clean_data( 

80 val_raw, 

81 impute_medians=fitted_clean_params["impute_medians"], 

82 outlier_bounds=fitted_clean_params["outlier_bounds"], 

83 ) 

84 test_clean, _, _ = clean_data( 

85 test_raw, 

86 impute_medians=fitted_clean_params["impute_medians"], 

87 outlier_bounds=fitted_clean_params["outlier_bounds"], 

88 ) 

89 report["cleaning"] = cleaning_report 

90 

91 # ---- 5. Feature engineering: fit scaler on train only, reuse on val/test ---- 

92 train_engineered, scaler, fe_report = engineer_features(train_clean) 

93 val_engineered, _, _ = engineer_features(val_clean, scaler=scaler) 

94 test_engineered, _, _ = engineer_features(test_clean, scaler=scaler) 

95 report["feature_engineering"] = fe_report 

96 

97 # ---- 6. Extract X/y for each split ---- 

98 X_train, y_train = extract_features_and_target(train_engineered) 

99 X_val, y_val = extract_features_and_target(val_engineered) 

100 X_test, y_test = extract_features_and_target(test_engineered) 

101 

102 # ---- 7. SMOTE -- train split only, applied AFTER the val/test splits 

103 # already have their final, untouched feature values ---- 

104 if apply_imbalance_handling: 

105 X_train, y_train, smote_report = apply_smote( 

106 X_train, y_train, sampling_strategy=smote_sampling_strategy 

107 ) 

108 report["smote"] = smote_report 

109 

110 return { 

111 "X_train": X_train, 

112 "y_train": y_train, 

113 "X_val": X_val, 

114 "y_val": y_val, 

115 "X_test": X_test, 

116 "y_test": y_test, 

117 "scaler": scaler, 

118 "report": report, 

119 } 

120

122def save_pipeline_report(report: dict[str, Any], output_path: str | Path) -> None: 

123 output_path = Path(output_path) 

124 output_path.parent.mkdir(parents=True, exist_ok=True) 

125 with open(output_path, "w") as f: 

126 json.dump(report, f, indent=2, default=str)

129def _main() -> None: 

130 parser = argparse.ArgumentParser(description="FraudShield AI — data preparation pipeline") 

131 parser.add_argument("--input", required=True, help="Path to the raw transactions CSV") 

132 parser.add_argument( 

133 "--report-output", 

134 default="ml_research/reports/data_preparation_report.json", 

135 help="Where to write the JSON pipeline report", 

136 ) 

137 parser.add_argument( 

138 "--scaler-output", 

139 default="app/ml/artifacts/amount_time_scaler.joblib", 

140 help="Where to persist the fitted RobustScaler", 

141 ) 

142 parser.add_argument( 

143 "--smote-sampling-strategy", 

144 type=float, 

145 default=0.5, 

146 help="Minority:majority ratio after SMOTE resampling", 

147 ) 

148 args = parser.parse_args() 

149 

150 result = prepare_data( 

151 args.input, smote_sampling_strategy=args.smote_sampling_strategy 

152 ) 

153 

154 save_pipeline_report(result["report"], args.report_output) 

155 

156 scaler_path = Path(args.scaler_output) 

157 scaler_path.parent.mkdir(parents=True, exist_ok=True) 

158 joblib.dump(result["scaler"], scaler_path) 

159 

160 print(f"Data preparation complete.") 

161 print(f" Train rows (post-SMOTE): {len(result['X_train'])}") 

162 print(f" Val rows: {len(result['X_val'])}") 

163 print(f" Test rows: {len(result['X_test'])}") 

164 print(f" Report written to: {args.report_output}") 

165 print(f" Scaler persisted to: {args.scaler_output}") 

166 

167 

168if __name__ == "__main__": 

169 sys.exit(_main() or 0)
