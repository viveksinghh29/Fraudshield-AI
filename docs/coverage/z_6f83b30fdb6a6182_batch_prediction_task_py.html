2Celery task: batch prediction over an already-uploaded batch of

26@celery_app.task(name="predict_batch_task", bind=True, max_retries=2) 

27def predict_batch_task(self, batch_id: str) -> dict: 

28 """ 

29 Entry point Celery calls. Delegates to an async implementation so 

30 the same repositories/services used by the synchronous API routes 

31 are reused here too, rather than duplicating query logic. 

32 

33 In production, a Celery worker process has no event loop already 

34 running, so asyncio.run() works directly. Celery's "eager" testing 

35 mode (task_always_eager=True) instead executes the task inline, 

36 synchronously, inside whatever process called .delay() -- which, 

37 for an async FastAPI route, means a loop is already running, and 

38 asyncio.run() raises "cannot be called from a running event loop". 

39 Detected exactly this way while adding API-level tests for the 

40 batch endpoints. Falls back to running the coroutine in a fresh 

41 thread (with its own new loop) only in that situation, so both the 

42 real worker path and the eager-mode test path work correctly. 

43 """ 

44 try: 

45 asyncio.get_running_loop() 

46 except RuntimeError: 

47 # No loop running in this thread -- the normal Celery worker case. 

48 return asyncio.run(_predict_batch_async(uuid.UUID(batch_id))) 

49 

50 import concurrent.futures 

51 

52 with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool: 

53 future = pool.submit(asyncio.run, _predict_batch_async(uuid.UUID(batch_id))) 

54 return future.result()

73 "Time": txn.time, 

74 "Amount": float(txn.amount),

88 risk_level=RiskLevel(result["risk_level"]), 

89 )

91 fraud_count += 1 

92 

93 await audit_repo.log( 

94 action="BATCH_PREDICTION_COMPLETED", 

95 resource_type="batch",

97 metadata={ 

98 "transaction_count": len(transactions), 

99 "fraud_count": fraud_count, 

100 "model_version": active_model_version.version_tag, 

101 }, 

102 ) 

103 await session.commit() 

104 

105 logger.info( 

106 "Batch prediction complete for batch_id=%s: %d transactions, %d fraud", 

107 batch_id, 

108 len(transactions),

110 )

112 return {
